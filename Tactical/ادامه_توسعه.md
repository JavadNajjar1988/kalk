#### الف) ایجاد Service Provider در `App.vue`:

```vue
<!-- App.vue -->
<template>
  <ServiceProvider v-if="services" v-bind="services">
    <router-view />
  </ServiceProvider>
  <div v-else class="loading">Loading...</div>
</template>

<script setup>
import { ref, onMounted, watch } from 'vue'
import { useRoute } from 'vue-router'
import { provideServices } from './composables/useServices'
import projectServices from './services/projectServices'
import projectListServices from './services/projectListServices'

const route = useRoute()
const services = ref(null)

// Load services based on route
watch(() => route.path, async (path) => {
  services.value = null
  
  if (path.startsWith('/project/')) {
    const projectId = route.params.id
    services.value = await projectServices(projectId)
  } else {
    services.value = await projectListServices()
  }
}, { immediate: true })

// Provide services when ready
watch(services, (newServices) => {
  if (newServices) {
    provideServices(newServices)
  }
})
</script>
```

### 4. **تبدیل Composables اضافی** 🟡

#### الف) useDebounce:

```javascript
// src/composables/useDebounce.js
import { ref, watch } from 'vue'

export function useDebounce(value, delay) {
  const debouncedValue = ref(value.value)
  
  let timeoutId = null
  watch(value, (newValue) => {
    clearTimeout(timeoutId)
    timeoutId = setTimeout(() => {
      debouncedValue.value = newValue
    }, delay)
  })
  
  return debouncedValue
}
```

#### ب) useList:

```javascript
// src/composables/useList.js
import { reactive } from 'vue'
import { multiselect } from '../model/selection/multiselect'
import { singleselect } from '../model/selection/singleselect'

export function useList(options = {}) {
  const strategy = options.multiselect ? multiselect : singleselect
  
  const state = reactive({
    entries: [],
    selected: [],
    focusIndex: -1,
    scroll: 'auto'
  })
  
  const dispatch = (event) => {
    const handler = strategy[event.type]
    if (handler) {
      Object.assign(state, handler(state, event))
    }
  }
  
  return [state, dispatch]
}
```

#### ج) useEmitter:

```javascript
// src/composables/useEmitter.js
import { computed } from 'vue'
import { useServices } from './useServices'
import Emitter from '../shared/emitter'

export function useEmitter(key) {
  const services = useServices()
  
  return computed(() => {
    services.emitters = services.emitters || {}
    services.emitters[key] = services.emitters[key] || new Emitter()
    return services.emitters[key]
  })
}
```

## 📚 منابع برای تبدیل

### React → Vue تبدیل:

1. **JSX → Template:**
   ```jsx
   // React
   <div className="test">{value}</div>
   
   // Vue
   <div class="test">{{ value }}</div>
   ```

2. **useState → ref:**
   ```javascript
   // React
   const [value, setValue] = useState(0)
   
   // Vue
   const value = ref(0)
   ```

3. **useEffect → onMounted/onUnmounted:**
   ```javascript
   // React
   useEffect(() => {
     // setup
     return () => {
       // cleanup
     }
   }, [])
   
   // Vue
   onMounted(() => {
     // setup
   })
   onUnmounted(() => {
     // cleanup
   })
   ```

4. **Context → Provide/Inject:**
   ```javascript
   // React
   const value = useContext(MyContext)
   
   // Vue
   const value = inject(MyContext)
   ```
