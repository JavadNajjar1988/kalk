### چالش 2: ساختار Feature با $ Property

**مشکل**: کد انتظار دارد feature یک property `$` داشته باشد که signals را نگه می‌دارد.

**راه حل**: ایجاد یک adapter:

```javascript
// adapter.js
import Signal from '@syncpoint/signal'

export function createFeatureAdapter(feature, options = {}) {
  const $ = {}
  
  // تبدیل properties به signals
  $.properties = Signal.of(feature.getProperties())
  $.geometry = Signal.of(feature.getGeometry())
  $.globalStyle = Signal.of(options.globalStyle || {})
  $.layerStyle = Signal.of(options.layerStyle || {})
  $.featureStyle = Signal.of(options.featureStyle || {})
  $.selectionMode = Signal.of(options.selectionMode || null)
  $.centerResolution = Signal.of(options.resolution || 1)
  
  // اضافه کردن $ به feature
  feature.$ = $
  
  return feature
}
```

### چالش 3: وابستگی به OpenLayers Geometry Types

**مشکل**: کد از انواع هندسه OpenLayers استفاده می‌کند.

**راه حل**: اگر از کتابخانه نقشه دیگری استفاده می‌کنید:

1. **برای Leaflet**: تبدیل به GeoJSON و سپس به OpenLayers
2. **برای Mapbox GL**: استفاده از `ol/format/GeoJSON` برای تبدیل
3. **برای Google Maps**: تبدیل به GeoJSON

```javascript
// converter.js
import { GeoJSON } from 'ol/format'

export function convertToOpenLayers(geometry, sourceFormat = 'geojson') {
  const format = new GeoJSON()
  return format.readGeometry(geometry)
}
```

### مرحله 3: ایجاد Adapter

```javascript
// adapters/featureAdapter.js
import Signal from '@syncpoint/signal'
import createStyles from '../style/styles'

export function adaptFeatureForStyling(feature, options = {}) {
  // ایجاد $ property
  const $ = {
    properties: Signal.of(feature.getProperties()),
    geometry: Signal.of(feature.getGeometry()),
    globalStyle: Signal.of(options.globalStyle || {}),
    layerStyle: Signal.of(options.layerStyle || {}),
    featureStyle: Signal.of(options.featureStyle || {}),
    selectionMode: Signal.of(options.selectionMode || null),
    centerResolution: Signal.of(options.resolution || 1)
  }
  
  feature.$ = $
  
  return feature
}

export function getStyles(feature, options = {}) {
  const adaptedFeature = adaptFeatureForStyling(feature, options)
  return createStyles(adaptedFeature)
}
```

### مرحله 4: استفاده در پروژه

```javascript
// example.js
import { getStyles } from './adapters/featureAdapter'
import { Style } from 'ol/style'

// ایجاد feature
const feature = new Feature({
  geometry: new Point([0, 0]),
  sidc: 'G*F*LT----'  // کد نماد
})

// دریافت استایل‌ها
const styleSignal = getStyles(feature, {
  resolution: map.getView().getResolution(),
  globalStyle: {
    'line-color': 'black',
    'line-width': 2
  }
})

// تبدیل signal به style (برای OpenLayers)
styleSignal.subscribe(styles => {
  const olStyles = styles.map(styleProps => {
    // تبدیل styleProps به ol/style/Style
    return new Style({
      stroke: styleProps.stroke ? new Stroke(styleProps.stroke) : undefined,
      fill: styleProps.fill ? new Fill(styleProps.fill) : undefined,
      image: styleProps.image,
      text: styleProps.text ? new Text(styleProps.text) : undefined
    })
  })
  
  feature.setStyle(olStyles.length === 1 ? olStyles[0] : olStyles)
})
```

## 🎯 نسخه ساده‌شده (بدون Signal)

اگر نمی‌خواهید از Signal استفاده کنید، می‌توانید یک نسخه ساده‌شده ایجاد کنید:

```javascript
// simpleStyles.js
import { createSymbolStyle } from './style/symbol'
import { createLineStringStyle } from './style/linestring'
import { createPolygonStyle } from './style/polygon'
import { createMultiPointStyle } from './style/multipoint'
import * as Geometry from './geometry/geometry'

export function createStyle(feature, options = {}) {
  const geometry = feature.getGeometry()
  const geometryType = Geometry.geometryType(geometry)
  const properties = feature.getProperties()
  const sidc = properties.sidc
  
  const styleOptions = {
    ...options,
    geometry,
    properties,
    sidc,
    resolution: options.resolution || 1
  }
  
  switch (geometryType) {
    case 'Point':
      return createSymbolStyle(styleOptions)
    case 'LineString':
      return createLineStringStyle(styleOptions)
    case 'Polygon':
      return createPolygonStyle(styleOptions)
    case 'MultiPoint':
      return createMultiPointStyle(styleOptions)
    default:
      return createDefaultStyle(styleOptions)
  }
}
```

این نسخه نیاز به بازنویسی توابع دارد اما ساده‌تر است.
