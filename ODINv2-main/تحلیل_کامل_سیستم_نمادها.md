# تحلیل کامل سیستم تولید و رسم نمادها در ODINv2

## مقدمه

ODINv2 یک سیستم اطلاعاتی فرماندهی و کنترل (C2IS) است که از استاندارد MIL-STD-2525C برای نمایش نمادهای نظامی استفاده می‌کند. این سیستم قابلیت رسم نمادهای مختلف برای انواع هندسه‌های جغرافیایی (Point، LineString، Polygon، MultiPoint) را دارد.

---

## معماری کلی سیستم

### 1. نقطه ورود اصلی: `styles.js`

فایل `src/renderer/ol/style/styles.js` نقطه ورود اصلی برای تولید استایل‌ها است:

```21:45:ODINv2-main/src/renderer/ol/style/styles.js
export default feature => {
  const { $ } = feature

  $.sidc = $.properties.map(R.prop('sidc'))
  $.parameterizedSIDC = $.sidc.map(parameterized)
  $.colorScheme = Signal.link(_colorScheme, [$.globalStyle, $.layerStyle, $.featureStyle])
  $.schemeStyle = Signal.link(_schemeStyle, [$.sidc, $.colorScheme])
  $.effectiveStyle = Signal.link(_effectiveStyle, [$.globalStyle, $.schemeStyle, $.layerStyle, $.featureStyle])
  $.styleRegistry = $.effectiveStyle
    .map(styleRegistry)
    .map(fn => xs => xs.map(fn))
  $.styleFactory = Signal.of(xs => xs.flatMap(styleFactory))

  const featureId = feature.getId()
  const geometryType = Geometry.geometryType(feature.getGeometry())

  if (ID.isMarkerId(featureId)) return marker($)
  else if (ID.isMeasureId(featureId)) return measure($)
  else if (geometryType === 'Point') return symbol($)
  else if (geometryType === 'Polygon') return polygon($)
  else if (geometryType === 'LineString') return linestring($)
  else if (geometryType === 'MultiPoint') return multipoint($)
  else if (geometryType === 'LineString:Point') return corridor($)
  else return fallback($)
}
```

- سیستم بر اساس نوع هندسه (`geometryType`) تصمیم می‌گیرد که از کدام ماژول استفاده کند
- از الگوی **Signal** برای مدیریت reactive state استفاده می‌شود
- SIDC (Symbol Identification Code) کد استاندارد نماد نظامی است

```124:161:ODINv2-main/src/renderer/ol/style/styleFactory.js
const makeSymbol = props => {
  const modes = { dark: 'Dark', medium: 'Medium', light: 'Light' }

  const fromEntries = entries => Object.fromEntries(entries)
  const entries = obj => Object.entries(obj)
  const rejectNil = R.reject(([, v]) => R.isNil(v))

    colorMode: modes[props['color-scheme']],
    monoColor: props['symbol-color'],
    outlineColor: props['symbol-halo-color'],
    outlineWidth: props['symbol-halo-width'],
    infoFields: props['symbol-text'],
    infoSize: props['symbol-text-size'],
    infoColor: props['symbol-text-color'],
    infoOutlineColor: props['symbol-text-halo-color'],
    infoOutlineWidth: props['symbol-text-halo-width'],
    strokeWidth: props['symbol-line-width'],
    fill: props['symbol-fill'],
    fillOpacity: props['symbol-fill-opacity'],
    frame: props['symbol-frame'],
    size: props['symbol-size'] || 60,
    ...props['symbol-modifiers']
  })

  const symbol = new Symbol(props['symbol-code'], { ...options, infoFields: true })
  const { width, height } = symbol.getSize()

  return Styles.icon({

    anchorYUnits: 'pixels',
    scale: props['icon-scale'] || 0.5
  })
```

```3:39:ODINv2-main/src/renderer/ol/style/polygon-styles/G_G_PY.js
export default ({ TS, PI_OVER_2, geometry }) => {
  const ring = geometry.getExteriorRing()
  const line = TS.lengthIndexedLine(ring)
  const envelope = ring.getEnvelopeInternal()
  const centroid = TS.centroid(ring)
  const [minX, maxX] = [envelope.getMinX(), envelope.getMaxX()]
  const [minY, maxY] = [envelope.getMinY(), envelope.getMaxY()]
  const length = Math.max(maxX - minX, maxY - minY)

  const PI_OVER_6 = PI_OVER_2 / 3

  const Xs = R.range(0, 12)
    .filter(i => i % 3 !== 0)
    .map(i => {
      const A = TS.projectCoordinate(centroid)([i * PI_OVER_6, length])
      const lineString = TS.lineString([centroid, A])
      const intersection = TS.intersection([geometry, lineString])
      const point = intersection.getPointN(1)
      const coord = TS.coordinate(point)
      const index = line.indexOf(coord)
      const segment = TS.segment(line.extractLine(index - 10, index + 10).getCoordinates())

      return {
        id: 'style:default-text',
        geometry: point,
        'text-field': '"X"',
        'text-padding': 5,
        'text-clipping': 'line',
        'text-rotate': TS.rotation(segment)
      }
    })

  return [
    { id: 'style:2525c/solid-stroke', geometry },
    ...Xs
  ]
}
```

```3:24:ODINv2-main/src/renderer/ol/style/polygon-styles/G_M_SP.js
export default ({ TS, PI_OVER_2, resolution, geometry }) => {
  const coordinates = geometry.getCoordinates()
  const lineString = TS.lineString(coordinates)
  const indexedLine = TS.lengthIndexedLine(lineString)
  const endIndex = indexedLine.getEndIndex()
  const delta = resolution * 20
  const segmentPoints = R.aperture(2, R.range(0, 1 + Math.ceil(endIndex / delta)).map(i => i * delta))

  const spikes = TS.collect(segmentPoints.map(([a, b]) => {
    const P1 = indexedLine.extractPoint((a + b) / 2)
    const [A, B] = [indexedLine.extractPoint(a), indexedLine.extractPoint(b)]
    const segment = TS.segment([A, B])
    const angle = segment.angle() - PI_OVER_2
    const P2 = TS.projectCoordinate(P1)([angle, delta * 0.75])
    return TS.lineString([P1, P2])
  }))

  return [
    { id: 'style:2525c/solid-stroke', geometry },
    { id: 'style:2525c/solid-stroke', geometry: spikes }
  ]
}
```

```3:37:ODINv2-main/src/renderer/ol/style/multipoint-styles/G_T_E.js
export default ({ TS, PI_OVER_2, PI, DEG2RAD, geometry }) => {
  const delta = 330 * DEG2RAD
  const coords = TS.coordinates(geometry)
  const segment = TS.segment(coords)
  const angle = segment.angle()
  const radius = segment.getLength()

  const quads = 64
  const arcs = [
    TS.arc(coords[0], radius, angle, delta, quads),
    TS.arc(coords[0], 0.8 * radius, angle, delta, quads)
  ]

  const teeth = R.range(1, arcs[0].length)
    .filter(i => i % 5 === 0)
    .map(i => [arcs[0][i - 1], arcs[1][i], arcs[0][i + 1]])
    .map(coords => TS.lineString(coords))

  const project = TS.projectCoordinates(radius, angle - delta + PI_OVER_2, R.last(arcs[0]))
  const arrow = project([[0.1, -0.1], [0, 0], [0.1, 0.1]])
  const path = TS.union([...teeth, TS.lineString(arcs[0]), TS.lineString(arrow)])
  const anchor = TS.point(arcs[0][Math.floor(arcs[0].length / 2)])
  const rotate = TS.rotation(segment) - PI / 12

  return [
    { id: 'style:2525c/default-stroke', geometry: path },
    {
      id: 'style:default-text',
      geometry: anchor,
      'text-field': '"I"',
      'text-rotate': rotate,
      'text-padding': 5
    }
  ]
}
```

## خلاصه روش‌های رسم

### 1. نمادهای نقطه‌ای:
- **روش**: تولید SVG از کتابخانه `@syncpoint/signs`
- **فرمت**: Data URI (inline SVG)
- **ویژگی**: ساده و مستقیم

### 2. نمادهای خطی:
- **روش**: 
  - ساده‌سازی و صاف‌سازی خط
  - تولید هندسه‌های اضافی (LineString، Polygon) بر اساس الگو
  - استفاده از `projectCoordinates` برای تبدیل مختصات نسبی
- **ویژگی**: پیچیده، نیاز به محاسبات هندسی دقیق

### 3. نمادهای چندوجهی:
- **روش**:
  - محاسبه centroid و envelope
  - تولید خطوط شعاعی یا spikeها
  - قرار دادن نمادها/متن‌ها در نقاط خاص
- **ویژگی**: نیاز به محاسبات تقاطع و placement

### 4. نمادهای چندنقطه‌ای:
- **روش**:
  - استفاده از دو نقطه برای تعیین مرکز و شعاع
  - تولید کمان‌ها و arcها
  - ترکیب چندین هندسه (teeth، arrows، arcs)
- **ویژگی**: شبیه به چندوجهی اما با هندسه دایره‌ای

2. **Reactive Programming**: استفاده از الگوی Signal برای مدیریت state
3. **Modular Design**: هر نوع نماد در فایل جداگانه تعریف شده
4. **Performance**: ساده‌سازی و clipping برای بهبود عملکرد
5. **Extensibility**: افزودن نماد جدید با ایجاد فایل جدید در پوشه مربوطه

## نتیجه‌گیری

سیستم تولید نمادها در ODINv2 یک سیستم پیچیده و قدرتمند است که:
- از استاندارد MIL-STD-2525C پشتیبانی می‌کند
- قابلیت رسم نمادهای پیچیده برای انواع هندسه‌ها را دارد
- از الگوریتم‌های پیشرفته هندسی استفاده می‌کند
- قابلیت توسعه و افزودن نمادهای جدید را دارد
